/** Globals injected into Kudium's asynchronous page hook callbacks. */
interface KudiumActivityInput {
  /** Required title, up to 120 characters. */
  title: string;
  /** Optional details, up to 2,000 characters. */
  message?: string;
  /** Optional HTTP(S) action link. */
  url?: string;
}
interface KudiumNoteInput {
  title: string;
  /** Markdown content. */
  body: string;
}
type KudiumPanelContent =
  | { type: 'text' | 'markdown'; text: string }
  | { type: 'json'; value: unknown }
  | {
      type: 'table';
      columns: string[];
      rows: Array<Array<string | number | boolean | null>>;
    }
  | { type: 'status'; state: 'loading' | 'success' | 'error'; text: string }
  | {
      type: 'html';
      html: string;
      css?: string;
      script?: string;
      data?: unknown;
      /** Explicit HTTP(S) image URLs; loaded without cookies or referrer. */
      images?: string[];
    };
interface KudiumPanelOutput {
  id: string;
  title: string;
  content: KudiumPanelContent;
  actions?: Array<{ id: string; label: string }>;
  reveal?: boolean;
}
interface KudiumHookContext {
  /** Cached macOS battery snapshot; unavailable devices return null fields. Preview uses fictional data. */
  readonly os: {
    readonly battery: {get():Promise<{available:boolean;percent:number|null;charging:boolean|null;onBattery:boolean|null;sampledAt:number|null;preview?:boolean}>};
    readonly power: {get():Promise<{onBattery:boolean;lowPowerMode:boolean|null}>;onChange(handler:(value:{onBattery:boolean;lowPowerMode:boolean|null})=>void|Promise<void>):()=>void};
    readonly appearance: {get():Promise<{dark:boolean;reducedMotion:boolean;reducedTransparency:boolean;highContrast:boolean}>;onChange(handler:(value:{dark:boolean;reducedMotion:boolean;reducedTransparency:boolean;highContrast:boolean})=>void|Promise<void>):()=>void};
    readonly network: {get():Promise<{online:boolean}>;onChange(handler:(value:{online:boolean})=>void|Promise<void>):()=>void};
    readonly locale: {get():Promise<{language:string;languages:string[];timeZone:string}>};
  };
  /** Browser-owned toast. Plain text; 5s expiry, or 10s when an action is declared. */
  readonly toast: {
    show(input:{id:string;title:string;message:string;durationMs?:number;type?:'info'|'warn'|'success'|'error';tone?:'info'|'success'|'warning'|'error';action?:{id:string;label:string}}):Promise<{shown:boolean;preview?:boolean}>;
    onAction(id:string,handler:()=>void|Promise<void>):()=>void;
  };

  /** Current browser theme; read again after theme changes. Also available as panel CSS variables. */
  readonly theme: Readonly<{primary: string; frame: string; contrast: string}>;
  readonly panel: {
    show(output: KudiumPanelOutput): Promise<{ id: string }>;
    update(output: KudiumPanelOutput): Promise<{ id: string }>;
    close(id: string): Promise<void>;
    loadState(id: string): Promise<any>;
    saveState(id: string, value: any): Promise<void>;
    clearState(id: string): Promise<void>;
    onAction(
      id: string,
      handler: (payload: unknown) => unknown | Promise<unknown>,
    ): () => void;
  };
  /** Unique identity for this hook stage execution. */
  readonly runId: string;
  /** Aborted when disabled, replaced, navigated away, or timed out. */
  readonly signal: AbortSignal;
  /** Register cleanup for DOM changes, observers and other resources. */
  onCleanup(cleanup: () => void): void;
  setInterval(
    callback: (...args: any[]) => void,
    delay: number,
    ...args: any[]
  ): number;
  setTimeout(
    callback: (...args: any[]) => void,
    delay: number,
    ...args: any[]
  ): number;
  listen(
    target: EventTarget,
    type: string,
    callback: (event: Event) => void,
    options?: boolean | AddEventListenerOptions,
  ): void;
  readonly url: string;
  readonly document: Document;
  readonly content: string;
  readonly activity: {
    create(
      input: KudiumActivityInput,
    ): Promise<{ id: string; deepLink: string }>;
  };
  readonly notes: {
    create(input: KudiumNoteInput): Promise<{
      recordId: string;
      title: string;
      deepLink: string;
      activityId?: string;
      warning?: string;
    }>;
  };
  readonly page: {
    reload(): void;
    back(): void;
    forward(): void;
    stop(): void;
  };
  /** Write to the hook log. */
  log(...values: unknown[]): void;
  /** Requires a configured, approved notification binding. */
  notify(key: string, text: string): Promise<unknown>;
}

declare const activity: KudiumHookContext['activity'];

declare const panel: KudiumHookContext['panel'];

declare const log: KudiumHookContext['log'];

/** Context menu handlers receive content scoped to the user’s click. */
interface KudiumActionField {
  readonly key: string;
  readonly type: string;
  readonly label: string;
  readonly name: string;
  readonly id: string;
  readonly placeholder: string;
  readonly autocomplete: string;
  readonly required: boolean;
  readonly maxLength?: number;
  readonly minLength?: number;
  readonly pattern?: string;
  /** Current text value; password values are excluded. */
  readonly value?: string;
}
interface KudiumActionContext {
  readonly apiVersion: 1;
  readonly url: string;
  readonly title: string;
  /** Never provided for a password field. */
  readonly selection?: string;
  /** Visible text, up to 100,000 characters. */
  readonly pageText?: string;
  readonly link?: string;
  /** Labels, rules and current text values from the clicked field. */
  readonly field?: { readonly type: string } & Partial<KudiumActionField>;
  /** Labels, rules and text values of up to 50 fields in the clicked form. */
  readonly form?: { readonly fields: ReadonlyArray<KudiumActionField> };
}
interface KudiumInputField {
  key: string;
  label?: string;
  type?:
    | 'text'
    | 'email'
    | 'tel'
    | 'url'
    | 'number'
    | 'date'
    | 'textarea'
    | 'password'
    | 'select';
  group?: string;
  required?: boolean;
  placeholder?: string;
  personalData?: string;
  options?: Array<{ value: string; label: string }>;
}
interface KudiumInputRequest {
  title?: string;
  fields: Array<string | KudiumInputField>;
}
/** Outputs are queued, validated, then applied by the browser after the handler ends. */
interface KudiumActionApi {
  readonly version: 1;
  showResult(text: string): void;
  readonly field: { fill(text: string): void };
  /** Choose a saved field locally and fill the clicked text input. No values returned. */
  /** Legacy request alias; use api.input.request with personalData field mappings. */
  readonly personalData: {
    /** Review and save the clicked form. Passwords remain private to the browser. */
    saveForm(): void;
    fill(): void;
    request(
      options: KudiumInputRequest,
    ): Promise<Readonly<Record<string, string>> | null>;
  };
  /** Grouped, typed fields with optional saved-data filling. Cancel returns null. */
  readonly input: {
    request(
      options: KudiumInputRequest,
    ): Promise<Readonly<Record<string, string>> | null>;
  };
  /** Keys come from context.form. Password fields are excluded. */
  readonly form: { fill(values: Array<{ key: string; value: string }>): void };
  /** Generates and fills directly; never returns the password to the handler. */
  readonly passwords: {
    generateAndFill(): void;
    /** Fill a custom password made by your script. */
    fill(text: string): void;
  };
  /** Opens the built-in AI composer. The user submits the request. */
  readonly ai: { compose(prompt: string): void };
  readonly notes: { create(input: KudiumNoteInput): void };
  readonly activity: {
    create(input: { title: string; message?: string }): void;
  };
}
